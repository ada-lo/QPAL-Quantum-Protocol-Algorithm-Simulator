// ============================================================
// 1D Discrete Quantum Walk — OpenQASM 3.0
// Coin qubit q[0], position register q[1..3] (positions 0-7).
// Walker starts at position 4 (binary 100).
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

// 1. Register Declaration
qubit[4] q;   // q[0]=coin, q[1..3]=position
bit[4] c;

// 2. Initialize position to |100⟩ (position 4)
x q[1];

// 3. Step 1 — Coin flip then conditional shift
h q[0];
// Right shift: controlled increment on position register
cx q[0], q[3];
ccx q[0], q[3], q[2];
ccx q[0], q[2], q[1];

// 4. Step 2 — Coin flip then shift again
h q[0];
cx q[0], q[3];
ccx q[0], q[3], q[2];
ccx q[0], q[2], q[1];

// 5. Measure full register
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
c[3] = measure q[3];
