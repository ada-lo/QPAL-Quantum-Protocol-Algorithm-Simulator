// ============================================================
// QEC: Three-Qubit Bit-Flip Repetition Code — OpenQASM 3.0
// Encodes |ψ⟩ as |ψ_L⟩ = α|000⟩+β|111⟩
// Detects and corrects a single X (bit-flip) error.
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

qubit[5] q;   // q[0]=logical, q[1..2]=copies, q[3..4]=syndrome ancilla
bit[2] syndrome;
bit[1] result;

// 1. Encode — spread logical qubit to 3 physical qubits
cx q[0], q[1];
cx q[0], q[2];

// 2. Simulate error — bit flip on q[1]
x q[1];

// 3. Syndrome measurement — detect which qubit flipped
cx q[0], q[3];
cx q[1], q[3];   // Compare q[0] vs q[1]
cx q[1], q[4];
cx q[2], q[4];   // Compare q[1] vs q[2]

syndrome[0] = measure q[3];
syndrome[1] = measure q[4];

// 4. Correction — syndrome 11 indicates q[1] flipped
if (syndrome[0] == 1 && syndrome[1] == 1) {
    x q[1];
}

// 5. Decode — reverse encoding
cx q[0], q[2];
cx q[0], q[1];

// 6. Measure logical qubit — should match original state
result[0] = measure q[0];
