// ============================================================
// Grover's Search Algorithm — OpenQASM 3.0
// Matches the custom pseudocode logic: 2-qubit search, target |11>.
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

qubit[2] q;
bit[2] c;

// Initialize search register
reset q[0];
reset q[1];

// Uniform superposition
h q[0];
h q[1];

// Oracle marks |11>
cz q[0], q[1];

// Diffusion operator
h q[0];
h q[1];
x q[0];
x q[1];
cz q[0], q[1];
x q[0];
x q[1];
h q[0];
h q[1];

// Readout
c[0] = measure q[0];
c[1] = measure q[1];
