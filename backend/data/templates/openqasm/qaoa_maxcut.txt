// ============================================================
// QAOA MaxCut — OpenQASM 3.0
// Matches the custom pseudocode logic for a 3-node triangle graph.
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

qubit[3] q;
bit[3] c;

reset q[0];
reset q[1];
reset q[2];

// Initial superposition
h q[0];
h q[1];
h q[2];

// Cost layer with gamma = 0.8
cx q[0], q[1];
rz(1.6) q[1];
cx q[0], q[1];

cx q[1], q[2];
rz(1.6) q[2];
cx q[1], q[2];

cx q[0], q[2];
rz(1.6) q[2];
cx q[0], q[2];

// Mixer layer with beta = 0.5
rx(1.0) q[0];
rx(1.0) q[1];
rx(1.0) q[2];

// Readout
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
