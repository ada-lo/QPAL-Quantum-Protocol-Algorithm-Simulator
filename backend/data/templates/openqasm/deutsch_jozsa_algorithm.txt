// ============================================================
// Deutsch-Jozsa Algorithm — OpenQASM 3.0
// Generalizes Deutsch: n-bit f(x) constant or balanced, 1 query.
// This example uses n=3 input qubits + 1 ancilla.
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

// 1. Register Declaration
qubit[4] q;   // q[0..2] = input qubits, q[3] = ancilla
bit[3] c;

// 2. Initialize ancilla to |1⟩
x q[3];

// 3. Hadamard on all qubits
h q[0];
h q[1];
h q[2];
h q[3];

// 4. Oracle — Balanced function example:
//    f(x) = x0 XOR x1 XOR x2
//    Each input qubit is CNOTed to ancilla
cx q[0], q[3];
cx q[1], q[3];
cx q[2], q[3];
//    For CONSTANT oracle: comment out all cx gates above.

// 5. Hadamard on input qubits only
h q[0];
h q[1];
h q[2];

// 6. Measure input register
//    All |0⟩ = constant; any |1⟩ = balanced
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
