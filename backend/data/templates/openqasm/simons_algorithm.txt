// ============================================================
// Simon's Algorithm — OpenQASM 3.0
// Finds hidden period s where f(x) = f(x XOR s).
// 2-qubit example with hidden string s = 11
// ============================================================
OPENQASM 3.0;
include "stdgates.inc";

// 1. Register Declaration: input + output registers (2 qubits each)
qubit[4] q;   // q[0..1] = input register, q[2..3] = output register
bit[2] c;

// 2. Hadamard on input register → uniform superposition
h q[0];
h q[1];

// 3. Oracle — Implements f(x) = f(x XOR s) for s = 11
//    Copies input to output (identity part)
cx q[0], q[2];
cx q[1], q[3];
//    XOR with s=11: adds extra path for x XOR 11
cx q[0], q[3];
cx q[1], q[2];

// 4. Hadamard on input register again
h q[0];
h q[1];

// 5. Measure input register
//    Results are strings y where y · s = 0 (mod 2)
//    Run classically multiple times to solve linear system for s
c[0] = measure q[0];
c[1] = measure q[1];
