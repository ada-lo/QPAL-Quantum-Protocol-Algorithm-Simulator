# ============================================================
# Teleportation Walkthrough — QuNetSim
# Matches the custom pseudocode logic with Alice sending a teleported state to Bob.
# ============================================================
from qunetsim.components import Host, Network
from qunetsim.objects import Qubit
import time


def main():
    network = Network.get_instance()
    network.start(["Alice", "Bob"])

    alice = Host("Alice")
    alice.add_connection("Bob")
    alice.start()

    bob = Host("Bob")
    bob.add_connection("Alice")
    bob.start()

    network.add_host(alice)
    network.add_host(bob)

    q = Qubit(alice)
    q.H()
    print("[Alice] Teleporting a |+> state to Bob...")
    alice.send_teleport("Bob", q)

    time.sleep(1)
    received = bob.get_qubit("Alice", wait=5)
    if received is not None:
        print(f"[Bob] Received teleported qubit. Measurement: {received.measure()}")
    else:
        print("[Bob] No teleported qubit received.")

    network.stop(stop_hosts=True)


if __name__ == "__main__":
    main()
